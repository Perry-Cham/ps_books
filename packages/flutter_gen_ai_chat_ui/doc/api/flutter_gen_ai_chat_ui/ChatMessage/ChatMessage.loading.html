Creates a loading placeholder that can later be replaced via ChatMessagesController.updateMessage with a rich widget or text.

// Show loading
controller.addMessage(ChatMessage.loading(
  user: aiUser,
  id: 'response-42',
  text: 'Searching for lawyers...',
));

// Later, replace with rich widget
controller.updateMessage(ChatMessage.rich(
  user: aiUser,
  id: 'response-42',
  resultKind: 'lawyer_search',
  data: results,
));

Implementation

factory ChatMessage.loading({
  required ChatUser user,
  required String id,
  String? text,
  String? loadingKind,
  DateTime? createdAt,
}) {
  return ChatMessage(
    text: text ?? '',
    user: user,
    createdAt: createdAt ?? DateTime.now(),
    customProperties: {
      'id': id,
      'isLoading': true,
      if (loadingKind != null) 'loadingKind': loadingKind,
    },
  );
}
